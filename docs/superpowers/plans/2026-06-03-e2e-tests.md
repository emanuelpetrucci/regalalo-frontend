# E2E Tests Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement Playwright e2e tests for the Regálalo frontend that exercise real features against the local backend, with fresh test state created and destroyed for every test.

**Architecture:** The backend already accepts `'dev-mock-recaptcha-token'` when `NODE_ENV !== 'production'`. The frontend's `RecaptchaService` returns that mock token when `recaptchaSiteKey` is empty. A new Angular `e2e` build config sets `recaptchaSiteKey: ''`, making the full auth flow work without real reCAPTCHA. Email verification is handled by writing directly to PostgreSQL from Playwright helpers (using `pg`). Each test that needs authentication creates a fresh pre-verified user via API + DB, injects the JWT and user data into `localStorage`, and deletes the user on cleanup — no shared state between tests.

**Tech Stack:** Playwright 1.54 (installed), @faker-js/faker (installed), pg (to install), Angular 21, Express backend on `localhost:3000`, Angular dev server on `localhost:4201`.

  ```typescript
  import { Pool } from 'pg';

  const DATABASE_URL =
    process.env['E2E_DATABASE_URL'] ?? 'postgresql://nolv:nolv@localhost:5432/nolv';

  let pool: Pool | null = null;

  function getPool(): Pool {
    if (!pool) {
      pool = new Pool({ connectionString: DATABASE_URL, max: 3 });
    }
    return pool;
  }

  export async function closeDb(): Promise<void> {
    if (pool) {
      await pool.end();
      pool = null;
    }
  }

  /** Mark a user's email as verified directly in the database. */
  export async function verifyUserEmail(userId: string): Promise<void> {
    await getPool().query(
      'UPDATE users SET email_verified = true, verification_token_hash = null WHERE id = $1',
      [userId],
    );
  }

  /** Delete a user and all their data (relies on ON DELETE CASCADE foreign keys). */
  export async function deleteUser(userId: string): Promise<void> {
    await getPool().query('DELETE FROM users WHERE id = $1', [userId]);
  }
  ```

  import type { APIRequestContext } from '@playwright/test';

  const API_URL = 'http://localhost:3000/api';
  const MOCK_RECAPTCHA_TOKEN = 'dev-mock-recaptcha-token';

  export async function apiRegister(
    request: APIRequestContext,
    user: {
      email: string;
      password: string;
      firstName: string;
      lastName: string;
      postalCode: string;
    },
  ): Promise<{ token: string }> {
    const res = await request.post(`${API_URL}/auth/register`, {
      data: { ...user, recaptchaToken: MOCK_RECAPTCHA_TOKEN },
    });
    if (!res.ok()) {
      throw new Error(`Register failed ${res.status()}: ${await res.text()}`);
    }
    return res.json() as Promise<{ token: string }>;
  }

  export async function apiLogin(
    request: APIRequestContext,
    email: string,
    password: string,
  ): Promise<{ token: string }> {
    const res = await request.post(`${API_URL}/auth/login`, {
      data: { email, password, recaptchaToken: MOCK_RECAPTCHA_TOKEN },
    });
    if (!res.ok()) {
      throw new Error(`Login failed ${res.status()}: ${await res.text()}`);
    }
    return res.json() as Promise<{ token: string }>;
  }

  export interface ApiProfile {
    id: string;
    email: string;
    firstName: string;
    lastName: string;
    emailVerified: boolean;
    reputationScore: number;
    postalCode: string;
    totalDonated: number;
    totalReceived: number;
  }

  export async function apiGetMe(
    request: APIRequestContext,
    token: string,
  ): Promise<ApiProfile> {
    const res = await request.get(`${API_URL}/users/me`, {
      headers: { Authorization: `Bearer ${token}` },
    });
    if (!res.ok()) {
      throw new Error(`GET /users/me failed ${res.status()}: ${await res.text()}`);
    }
    return res.json() as Promise<ApiProfile>;
  }

  export async function apiPublishItem(
    request: APIRequestContext,
    token: string,
    item: {
      title: string;
      description: string;
      category: string;
      condition: string;
      postalCode: string;
    },
  ): Promise<{ id: string }> {
    const res = await request.post(`${API_URL}/items`, {
      headers: { Authorization: `Bearer ${token}` },
      multipart: { ...item },
    });
    if (!res.ok()) {
      throw new Error(`Publish item failed ${res.status()}: ${await res.text()}`);
    }
    return res.json() as Promise<{ id: string }>;
  }

- [ ] **Step 2: Create e2e/helpers/auth.ts**

  ```typescript
  import { faker } from '@faker-js/faker';
  import type { APIRequestContext, BrowserContext } from '@playwright/test';
  import { apiRegister, apiLogin, apiGetMe, type ApiProfile } from './api';
  import { verifyUserEmail, deleteUser } from './db';

  export interface TestSession {
    token: string;
    user: ApiProfile;
    password: string;
  }

  const TOKEN_KEY = 'rgl_auth_token';
  const USER_KEY = 'rgl_auth_user';

  export function generateTestUser() {
    return {
      email: `e2e-${faker.string.alphanumeric(10)}@example.com`,
      password: 'Test1234!',
      firstName: faker.person.firstName(),
      lastName: faker.person.lastName(),
      postalCode: '28001',
    };
  }

  /** Registers a user via API, marks email as verified in DB, and logs in. */
  export async function createVerifiedUser(request: APIRequestContext): Promise<TestSession> {
    const userData = generateTestUser();
    const { token: registerToken } = await apiRegister(request, userData);

    // Decode JWT payload (base64url) to get the user ID
    const payloadB64 = registerToken.split('.')[1];
    const payload = JSON.parse(
      Buffer.from(payloadB64, 'base64url').toString('utf-8'),
    ) as { sub: string };

    await verifyUserEmail(payload.sub);

    // Log in again so the JWT reflects the current state
    const { token } = await apiLogin(request, userData.email, userData.password);
    const profile = await apiGetMe(request, token);

    return { token, user: profile, password: userData.password };
  }

  /** Injects JWT and user data into the browser context via localStorage init script. */
  export async function injectSession(
    context: BrowserContext,
    session: TestSession,
  ): Promise<void> {
    await context.addInitScript(
      ({ tokenKey, userKey, token, user }) => {
        localStorage.setItem(tokenKey, token);
        localStorage.setItem(userKey, JSON.stringify(user));
      },
      { tokenKey: TOKEN_KEY, userKey: USER_KEY, token: session.token, user: session.user },
    );
  }

  /** Deletes a test user and all their cascade-deleted data. */
  export async function cleanupUser(userId: string): Promise<void> {
    await deleteUser(userId);
  }
  ```

## Notes on What This Plan Does Not Cover

These are candidates for a follow-up plan:

- **Delivery session flow** — requires two users (donor + receiver), item publication, request creation, and candidate selection before a delivery session exists. Too many setup steps for phase 1.
- **Full item publish with image upload** — Playwright can handle file inputs, but requires a real test image file and more complex step navigation.
- **Candidate selection dialog** — depends on having an item with requests, which requires a second test user making a request.
- **Password reset** — the email provider is `console` mode, so the reset token only appears in backend logs. Requires reading stdout or a test endpoint to capture the token.
- **Accessibility audits** — `@axe-core/playwright` is already installed; add `checkA11y(page)` to any spec to run an audit.
